CREATE TABLE IF NOT EXISTS public.direction_notification_deliveries (
 key text PRIMARY KEY CHECK(length(key)<=320),
 status text NOT NULL DEFAULT 'reserved' CHECK(status IN('reserved','sent','uncertain')),
 provider_message_id text,created_at timestamptz NOT NULL DEFAULT now(),updated_at timestamptz NOT NULL DEFAULT now()
);
ALTER TABLE public.direction_notification_deliveries ENABLE ROW LEVEL SECURITY;
REVOKE ALL ON public.direction_notification_deliveries FROM PUBLIC,anon,authenticated;
GRANT SELECT,INSERT,UPDATE ON public.direction_notification_deliveries TO service_role;
